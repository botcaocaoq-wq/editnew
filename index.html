class MotionFXEngine { constructor() { this.canvas = document.getElementById('videoCanvas'); this.ctx = this.canvas.getContext('2d'); this.bgImage = null; this.currentFX = 'none'; this.isPinMode = false; this.pins = []; this.time = 0; this.speed = 1; this.amount = 30; this.initEvents(); this.animate(); } initEvents() { document.getElementById('slider-speed').addEventListener('input', (e) => this.speed = parseFloat(e.target.value)); document.getElementById('slider-amount').addEventListener('input', (e) => this.amount = parseInt(e.target.value)); const btnPin = document.getElementById('btn-pin-mode'); const btnClearPins = document.getElementById('btn-clear-pins'); const fxModes = ['none', 'stretch', 'wave', 'wavewarp', 'bend']; btnPin?.addEventListener('click', () => { this.isPinMode = !this.isPinMode; btnPin.classList.toggle('active', this.isPinMode); btnPin.style.background = this.isPinMode ? '#ca8a04' : '#eab308'; }); btnClearPins?.addEventListener('click', () => { this.pins = []; }); fxModes.forEach(mode => { document.getElementById(`btn-fx-${mode}`)?.addEventListener('click', (e) => { this.isPinMode = false; btnPin.classList.remove('active'); btnPin.style.background = '#eab308'; fxModes.forEach(m => document.getElementById(`btn-fx-${m}`)?.classList.remove('active')); e.target.classList.add('active'); this.currentFX = mode; }); }); const btnUpload = document.getElementById('btn-upload'); const fileInput = document.getElementById('image-upload'); btnUpload?.addEventListener('click', () => fileInput.click()); fileInput?.addEventListener('change', (e) => { const files = e.target.files; if (!files || files.length === 0) return; const reader = new FileReader(); reader.onload = (ev) => { const img = new Image(); img.onload = () => { this.bgImage = img; this.pins = []; const maxBoundingWidth = 800; const maxBoundingHeight = 450; let newWidth = img.width; let newHeight = img.height; if (img.width > maxBoundingWidth || img.height > maxBoundingHeight) { const ratioX = maxBoundingWidth / img.width; const ratioY = maxBoundingHeight / img.height; const scale = Math.min(ratioX, ratioY); newWidth = img.width * scale; newHeight = img.height * scale; } this.canvas.width = newWidth; this.canvas.height = newHeight; this.currentFX = 'none'; fxModes.forEach(m => document.getElementById(`btn-fx-${m}`)?.classList.remove('active')); document.getElementById('btn-fx-none')?.classList.add('active'); }; img.src = ev.target.result; }; reader.readAsDataURL(files[0]); }); this.canvas.addEventListener('mousedown', (e) => { const rect = this.canvas.getBoundingClientRect(); const mousePos = { x: e.clientX - rect.left, y: e.clientY - rect.top }; if (this.isPinMode) { this.pins.push(mousePos); } }); } animate() { this.time += 0.05 * this.speed; this.renderFX(); requestAnimationFrame(() => this.animate()); } renderFX() { const w = this.canvas.width; const h = this.canvas.height; if (!this.bgImage) { this.ctx.fillStyle = '#18181b'; this.ctx.fillRect(0, 0, w, h); return; } this.ctx.drawImage(this.bgImage, 0, 0, w, h); const srcData = this.ctx.getImageData(0, 0, w, h); const dstData = this.ctx.createImageData(w, h); const src = srcData.data; const dst = dstData.data; const pinRadius = 40; for (let y = 0; y < h; y++) { for (let x = 0; x < w; x++) { let pinFactor = 1.0; for (let p of this.pins) { let dx = x - p.x; let dy = y - p.y; let dist = Math.sqrt(dx * dx + dy * dy); if (dist < pinRadius) { let factor = dist / pinRadius; if (factor < pinFactor) pinFactor = factor; } } let sampleX = x; let sampleY = y; if (this.currentFX !== 'none') { let offsetX = 0; let offsetY = 0; switch (this.currentFX) { case 'stretch': offsetY = Math.sin(x * 0.2 + this.time) * (this.amount * 0.2); break; case 'wave': offsetX = Math.sin(y * 0.05 + this.time) * (this.amount * 0.3); break; case 'wavewarp': offsetX = Math.sin((x + y) * 0.04 + this.time * 1.5) * (this.amount * 0.4); offsetY = Math.cos((x - y) * 0.04 + this.time * 1.5) * (this.amount * 0.2); break; case 'bend': offsetX = Math.sin(y / h * Math.PI + this.time) * (this.amount * 0.5); break; } sampleX += offsetX * pinFactor; sampleY += offsetY * pinFactor; } sampleX = Math.max(0, Math.min(w - 1, Math.floor(sampleX))); sampleY = Math.max(0, Math.min(h - 1, Math.floor(sampleY))); const srcIdx = (sampleY * w + sampleX) * 4; const dstIdx = (y * w + x) * 4; dst[dstIdx] = src[srcIdx]; dst[dstIdx + 1] = src[srcIdx + 1]; dst[dstIdx + 2] = src[srcIdx + 2]; dst[dstIdx + 3] = src[srcIdx + 3]; } } this.ctx.putImageData(dstData, 0, 0); this.pins.forEach(p => { this.ctx.beginPath(); this.ctx.arc(p.x, p.y, 5, 0, 2 * Math.PI); this.ctx.fillStyle = '#ef4444'; this.ctx.fill(); this.ctx.strokeStyle = '#ffffff'; this.ctx.lineWidth = 2; this.ctx.stroke(); }); } } window.addEventListener('DOMContentLoaded', () => { new MotionFXEngine(); });
